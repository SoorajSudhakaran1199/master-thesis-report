% -------------------------
% HEADING FORMATTING (KOMA-safe + fallback)
% - If you use scrreprt (KOMA), do NOT rely only on titlesec.
% - This block automatically chooses the right method.
% -------------------------
\makeatletter
\@ifclassloaded{scrreprt}{%
  % ===== KOMA-Script way (recommended if scrreprt) =====
  \setkomafont{chapter}{\bfseries\fontsize{14}{16}\selectfont}
  \setkomafont{section}{\bfseries\fontsize{12}{14}\selectfont}
  \setkomafont{subsection}{\bfseries\fontsize{11}{13}\selectfont}
}{%
  % ===== Standard report/book way =====
  \usepackage{titlesec}

  % Chapter (14 pt)
  \titleformat{\chapter}
    {\normalfont\bfseries\fontsize{14}{16}\selectfont}
    {\thechapter}{1em}{}

  % Section (12 pt)
  \titleformat{\section}
    {\normalfont\bfseries\fontsize{12}{14}\selectfont}
    {\thesection}{1em}{}

  % Subsection (11 pt)
  \titleformat{\subsection}
    {\normalfont\bfseries\fontsize{11}{13}\selectfont}
    {\thesubsection}{1em}{}

  % Chapter spacing (moderate)
  \titlespacing*{\chapter}{0pt}{20pt}{20pt}
}%
\makeatother
